cmake_minimum_required(VERSION 3.24)

project(SimNodus VERSION 0.0.0 DESCRIPTION "Mixed-signal and embedded systems simulator" LANGUAGES CXX)

# This foundation does not build a simulator or download runtime dependencies.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(simnodus_build_options INTERFACE)
add_library(simnodus::build_options ALIAS simnodus_build_options)
target_compile_features(simnodus_build_options INTERFACE cxx_std_20)

find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_custom_target(check-repository
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_repository.py"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    COMMENT "Checking repository documentation and structure"
    VERBATIM
)

include(CTest)
if(BUILD_TESTING)
    add_executable(backend_contract_tests tests/headless/backend_contracts.cpp)
    target_include_directories(backend_contract_tests PRIVATE src)
    target_link_libraries(backend_contract_tests PRIVATE simnodus::build_options)
    add_test(NAME backend-contracts COMMAND backend_contract_tests)
endif()
option(SIMNODUS_BUILD_RC_FIXTURE "Build the bounded Windows ngspice fixture runner" OFF)
if(SIMNODUS_BUILD_RC_FIXTURE)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The extracted RC fixture requires Windows x64.")
    endif()
    set(NGSPICE_ROOT "" CACHE PATH "Explicit verified Spice64_dll package")
    if(NOT EXISTS "${NGSPICE_ROOT}/include/ngspice/sharedspice.h")
        message(FATAL_ERROR "Set NGSPICE_ROOT to the verified DLL package.")
    endif()
    add_library(simnodus_rc_adapter STATIC src/adapters/ngspice/rc_fixture.cpp)
    target_include_directories(simnodus_rc_adapter PRIVATE src tools/backend_probe "${NGSPICE_ROOT}/include")
    target_compile_definitions(simnodus_rc_adapter PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(simnodus_rc_adapter PRIVATE simnodus::build_options)
    add_executable(simnodus_rc_fixture apps/cli/rc_fixture_main.cpp)
    target_include_directories(simnodus_rc_fixture PRIVATE src)
    target_link_libraries(simnodus_rc_fixture PRIVATE simnodus_rc_adapter)
    if(MSVC)
        target_compile_options(simnodus_rc_adapter PRIVATE /W4 /WX /permissive- /EHsc)
    endif()
endif()
