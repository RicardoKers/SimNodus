cmake_minimum_required(VERSION 3.24)
project(SimNodusRenodeClientExperiment LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "SN-019 currently requires native Windows x64.")
endif()
set(PREPARED_ROOT "" CACHE PATH "Verified output of prepare.py")
if(NOT EXISTS "${PREPARED_ROOT}/renode_api.c")
    message(FATAL_ERROR "Run prepare.py explicitly and set PREPARED_ROOT.")
endif()
foreach(variant IN ITEMS normal fragmented)
    add_library(client_${variant} STATIC "${PREPARED_ROOT}/renode_api.c" transport.c)
    target_include_directories(client_${variant} PUBLIC "${PREPARED_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
    set_target_properties(client_${variant} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    target_compile_definitions(client_${variant} PRIVATE WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(client_${variant} PUBLIC ws2_32)
    add_executable(probe_${variant} probe.cpp)
    target_compile_features(probe_${variant} PRIVATE cxx_std_20)
    target_compile_definitions(probe_${variant} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(probe_${variant} PRIVATE client_${variant})
    if(variant STREQUAL "fragmented")
        target_compile_definitions(client_${variant} PUBLIC SN_TRANSFER_LIMIT=1)
    endif()
    if(MSVC)
        target_compile_options(client_${variant} PRIVATE /W3 /WX)
        target_compile_options(probe_${variant} PRIVATE /W4 /WX /permissive-)
    endif()
endforeach()
