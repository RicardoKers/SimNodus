cmake_minimum_required(VERSION 3.24)
project(SimNodusE01 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "E-01 currently requires native Windows x64.")
endif()
set(NGSPICE_ROOT "" CACHE PATH "Verified Spice64_dll directory")
if(NOT EXISTS "${NGSPICE_ROOT}/include/ngspice/sharedspice.h")
    message(FATAL_ERROR "Set NGSPICE_ROOT to the verified DLL package.")
endif()
add_executable(e01 e01.cpp)
target_compile_features(e01 PRIVATE cxx_std_20)
set_target_properties(e01 PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(e01 PRIVATE "${NGSPICE_ROOT}/include" "../../../tools/backend_probe")
target_compile_definitions(e01 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(e01 PRIVATE /W4 /WX /permissive-)
endif()
