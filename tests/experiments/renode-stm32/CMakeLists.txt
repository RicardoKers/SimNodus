cmake_minimum_required(VERSION 3.24)
project(SimNodusSTM32Experiment LANGUAGES C CXX)
set(FIRMWARE_ROOT "" CACHE PATH "Output of build_firmware.py")
if(NOT EXISTS "${FIRMWARE_ROOT}/firmware_layout.h")
    message(FATAL_ERROR "Build the owned firmware explicitly and set FIRMWARE_ROOT.")
endif()
add_subdirectory(../renode-client "${CMAKE_CURRENT_BINARY_DIR}/client")
add_executable(stm32_probe probe.cpp)
target_compile_features(stm32_probe PRIVATE cxx_std_20)
target_include_directories(stm32_probe PRIVATE "${FIRMWARE_ROOT}")
target_link_libraries(stm32_probe PRIVATE client_normal)
if(MSVC)
    target_compile_options(stm32_probe PRIVATE /W4 /WX /permissive-)
endif()
