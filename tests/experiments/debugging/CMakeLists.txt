cmake_minimum_required(VERSION 3.24)
project(SimNodusE05 LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "E-05 currently requires native Windows x64.")
endif()

set(PREPARED_ROOT "" CACHE PATH "Verified output of the SN-019 preparation step")
set(NGSPICE_ROOT "" CACHE PATH "Verified Spice64_dll directory")
if(NOT EXISTS "${NGSPICE_ROOT}/include/ngspice/sharedspice.h")
    message(FATAL_ERROR "Set NGSPICE_ROOT to the verified DLL package.")
endif()
add_subdirectory(../renode-client "${CMAKE_CURRENT_BINARY_DIR}/client")

add_executable(e05_control control.cpp)
target_compile_features(e05_control PRIVATE cxx_std_20)
set_target_properties(e05_control PROPERTIES CXX_EXTENSIONS OFF)
target_compile_definitions(e05_control PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(e05_control PRIVATE client_normal)
target_compile_definitions(client_normal PRIVATE SN_TIMEOUT_MS=30000)

add_executable(e05_circuit circuit.cpp)
target_compile_features(e05_circuit PRIVATE cxx_std_20)
set_target_properties(e05_circuit PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(e05_circuit PRIVATE "${NGSPICE_ROOT}/include" "../../../tools/backend_probe")
target_compile_definitions(e05_circuit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)

if(MSVC)
    target_compile_options(e05_control PRIVATE /W4 /WX /permissive- /EHsc)
    target_compile_options(e05_circuit PRIVATE /W4 /WX /permissive- /EHsc)
endif()

add_executable(e05_persistent_circuit persistent_circuit.cpp)
target_compile_features(e05_persistent_circuit PRIVATE cxx_std_20)
set_target_properties(e05_persistent_circuit PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(e05_persistent_circuit PRIVATE "${NGSPICE_ROOT}/include" "../../../tools/backend_probe")
target_compile_definitions(e05_persistent_circuit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(e05_persistent_circuit PRIVATE /W4 /WX /permissive- /EHsc)
endif()

add_executable(e05_joint_circuit joint_circuit.cpp)
target_compile_features(e05_joint_circuit PRIVATE cxx_std_20)
set_target_properties(e05_joint_circuit PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(e05_joint_circuit PRIVATE "${NGSPICE_ROOT}/include" "../../../tools/backend_probe")
target_compile_definitions(e05_joint_circuit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
    target_compile_options(e05_joint_circuit PRIVATE /W4 /WX /permissive- /EHsc)
endif()
